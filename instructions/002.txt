# Step 2: Team Management & Collaboration

## Overview
This step implements comprehensive team management features including roster management, availability tracking, task assignments, and shared calendar integration with modern UI design trends.

## Design Philosophy
Based on 2024-2025 iOS design trends, we're implementing:
- **Glassmorphism**: Subtle transparency effects for depth
- **Neumorphism**: Soft, tactile button designs
- **Dynamic Color**: Adaptive color schemes that respond to system settings
- **Micro-interactions**: Smooth animations and haptic feedback
- **Card-based layouts**: Clean, organized information hierarchy
- **Gradient accents**: Modern color transitions for visual appeal

```swift
        case .meeting: return "green"
        case .outreach: return "purple"
        }
    }

enum EventStatus: String, Codable, CaseIterable {
    case scheduled = "scheduled"
    case inProgress = "in_progress"
    case completed = "completed"
    case cancelled = "cancelled"
    
    var displayName: String {
        switch self {
        case .scheduled: return "Scheduled"
        case .inProgress: return "In Progress"
        case .completed: return "Completed"
        case .cancelled: return "Cancelled"
        }
    }

struct Event: Codable, Identifiable, Hashable {
    let id: UUID
    let teamId: UUID
    let title: String
    let description: String?
    let startDate: Date
    let endDate: Date?
    let location: String?
    let eventType: EventType
    let status: EventStatus
    let googleCalendarId: String?
    let createdAt: Date
    let updatedAt: Date
    
    enum CodingKeys: String, CodingKey {
        case id, title, description, location
        case teamId = "team_id"
        case startDate = "start_date"
        case endDate = "end_date"
        case eventType = "event_type"
        case status
        case googleCalendarId = "google_calendar_id"
        case createdAt = "created_at"
        case updatedAt = "updated_at"
    }
    
    var duration: TimeInterval {
        let end = endDate ?? startDate.addingTimeInterval(3600) // Default 1 hour
        return end.timeIntervalSince(startDate)
    }
    
    var isAllDay: Bool {
        return duration >= 86400 // 24 hours
    }
}
```

```swift
    case medium = 2
    case high = 3

        case .low: return "Low"

        case .medium: return "yellow"
        case .high: return "orange"
        case .urgent: return "red"
        }
    }
}

struct Task: Codable, Identifiable, Hashable {
    let id: UUID
    let teamId: UUID
    let title: String
    let description: String?
    let status: TaskStatus
    let priority: TaskPriority
    let assigneeId: UUID?
    let dueDate: Date?
    let createdBy: UUID
    let createdAt: Date
    let updatedAt: Date
    
    enum CodingKeys: String, CodingKey {
        case id, title, description, status, priority, dueDate
        case teamId = "team_id"
        case assigneeId = "assignee_id"
        case createdBy = "created_by"
        case createdAt = "created_at"
        case updatedAt = "updated_at"
    }
    
    var isOverdue: Bool {
        guard let dueDate = dueDate else { return false }
        return dueDate < Date() && status != .done
    }
    
    var daysUntilDue: Int? {
        guard let dueDate = dueDate else { return nil }
        return Calendar.current.dateComponents([.day], from: Date(), to: dueDate).day
    }
}
```

### 2.4 Create TeamService
Create `Services/TeamService.swift`:

```swift
import Foundation
import Supabase
import Combine

@MainActor
class TeamService: ObservableObject {
    @Published var currentTeam: Team?
    @Published var teamMembers: [Profile] = []
    @Published var events: [Event] = []
    @Published var tasks: [Task] = []
    @Published var isLoading = false
    @Published var errorMessage: String?
    
    private let supabase = SupabaseService.shared.client
    
    // MARK: - Team Management
    func createTeam(name: String, number: String, school: String?, city: String?, state: String?) async throws {
        isLoading = true
        errorMessage = nil
        defer { isLoading = false }
        
        do {
            let team = Team(
                id: UUID(),
                name: name,
                number: number,
                school: school,
                city: city,
                state: state,
                country: "USA",
                createdAt: Date(),
                updatedAt: Date()
            )
            
            try await supabase.database
                .from("teams")
                .insert(team)
                .execute()
            
            currentTeam = team
        } catch {
            errorMessage = error.localizedDescription
            throw error
        }
    }
    
    func joinTeam(teamNumber: String) async throws {
        isLoading = true
        errorMessage = nil
        defer { isLoading = false }
        
        do {
            let team: Team = try await supabase.database
                .from("teams")
                .select()
                .eq("number", value: teamNumber)
                .single()
                .execute()
                .value
            
            // Update user's team_id
            try await supabase.database
                .from("profiles")
                .update(["team_id": team.id])
                .eq("id", value: supabase.auth.currentUser?.id)
                .execute()
            
            currentTeam = team
            await loadTeamData()
        } catch {
            errorMessage = error.localizedDescription
            throw error
        }
    }
    
    func loadTeamData() async {
        guard let teamId = currentTeam?.id else { return }
        
        do {
            // Load team members
            let members: [Profile] = try await supabase.database
                .from("profiles")
                .select()
                .eq("team_id", value: teamId)
                .execute()
                .value
            
            teamMembers = members
            
            // Load events
            let teamEvents: [Event] = try await supabase.database
                .from("events")
                .select()
                .eq("team_id", value: teamId)
                .order("start_date", ascending: true)
                .execute()
                .value
            
            events = teamEvents
            
            // Load tasks
            let teamTasks: [Task] = try await supabase.database
                .from("tasks")
                .select()
                .eq("team_id", value: teamId)
                .order("created_at", ascending: false)
                .execute()
                .value
            
            tasks = teamTasks
        } catch {
            print("Failed to load team data: \(error)")
            errorMessage = "Failed to load team data"
        }
    }
    
    // MARK: - Event Management
    func createEvent(title: String, description: String?, startDate: Date, endDate: Date?, location: String?, eventType: EventType) async throws {
        guard let teamId = currentTeam?.id else { throw TeamError.noTeamSelected }
        
        let event = Event(
            id: UUID(),
            teamId: teamId,
            title: title,
            description: description,
            startDate: startDate,
            endDate: endDate,
            location: location,
            eventType: eventType,
            status: .scheduled,
            googleCalendarId: nil,
            createdAt: Date(),
            updatedAt: Date()
        )
        
        try await supabase.database
            .from("events")
            .insert(event)
            .execute()
        
        await loadTeamData()
    }
    
    func updateEvent(_ event: Event) async throws {
        try await supabase.database
            .from("events")
            .update(event)
            .eq("id", value: event.id)
            .execute()
        
        await loadTeamData()
    }
    
    func deleteEvent(_ event: Event) async throws {
        try await supabase.database
            .from("events")
            .delete()
            .eq("id", value: event.id)
            .execute()
        
        await loadTeamData()
    }
    
    // MARK: - Task Management
    func createTask(title: String, description: String?, priority: TaskPriority, assigneeId: UUID?, dueDate: Date?) async throws {
        guard let teamId = currentTeam?.id,
              let currentUserId = supabase.auth.currentUser?.id else { throw TeamError.noTeamSelected }
        
        let task = Task(
            id: UUID(),
            teamId: teamId,
            title: title,
            description: description,
            status: .todo,
            priority: priority,
            assigneeId: assigneeId,
            dueDate: dueDate,
            createdBy: currentUserId,
            createdAt: Date(),
            updatedAt: Date()
        )
        
        try await supabase.database
            .from("tasks")
            .insert(task)
            .execute()
        
        await loadTeamData()
    }
    
    func updateTask(_ task: Task) async throws {
        try await supabase.database
            .from("tasks")
            .update(task)
            .eq("id", value: task.id)
            .execute()
        
        await loadTeamData()
    }
    
    func deleteTask(_ task: Task) async throws {
        try await supabase.database
            .from("tasks")
            .delete()
            .eq("id", value: task.id)
            .execute()
        
        await loadTeamData()
    }
}

enum TeamError: LocalizedError {
    case noTeamSelected
    case teamNotFound
    case invalidTeamNumber
    
    var errorDescription: String? {
        switch self {
        case .noTeamSelected:
            return "No team selected"
        case .teamNotFound:
            return "Team not found"
        case .invalidTeamNumber:
            return "Invalid team number"
        }
    }
}
```

```swift
import SwiftUI

struct DesignSystem {
    // MARK: - Colors
    struct Colors {
        static let primary = Color.blue
        static let secondary = Color.purple
        static let accent = Color.orange
        static let success = Color.green
        static let warning = Color.yellow
        static let error = Color.red
        static let background = Color(.systemBackground)
        static let secondaryBackground = Color(.secondarySystemBackground)
        static let tertiaryBackground = Color(.tertiarySystemBackground)
    }
    
    // MARK: - Gradients
    struct Gradients {
        static let primary = LinearGradient(
            colors: [Colors.primary, Colors.secondary],
            startPoint: .topLeading,
            endPoint: .bottomTrailing
        )
        
        static let card = LinearGradient(
            colors: [Colors.background, Colors.secondaryBackground],
            startPoint: .top,
            endPoint: .bottom
        )
        
        static let glass = LinearGradient(
            colors: [Color.white.opacity(0.1), Color.white.opacity(0.05)],
            startPoint: .topLeading,
            endPoint: .bottomTrailing
        )
    }
    
    // MARK: - Spacing
    struct Spacing {
        static let xs: CGFloat = 4
        static let sm: CGFloat = 8
        static let md: CGFloat = 16
        static let lg: CGFloat = 24
        static let xl: CGFloat = 32
        static let xxl: CGFloat = 48
    }
    
    // MARK: - Corner Radius
    struct CornerRadius {
        static let sm: CGFloat = 8
        static let md: CGFloat = 12
        static let lg: CGFloat = 16
        static let xl: CGFloat = 24
    }
    
    // MARK: - Shadows
    struct Shadows {
        static let sm = Shadow(color: .black.opacity(0.1), radius: 2, x: 0, y: 1)
        static let md = Shadow(color: .black.opacity(0.15), radius: 4, x: 0, y: 2)
        static let lg = Shadow(color: .black.opacity(0.2), radius: 8, x: 0, y: 4)
    }
}

struct Shadow {
    let color: Color
    let radius: CGFloat
    let x: CGFloat
    let y: CGFloat
}

// MARK: - View Extensions
extension View {
    func glassmorphism() -> some View {
        self
            .background(
                RoundedRectangle(cornerRadius: DesignSystem.CornerRadius.lg)
                    .fill(DesignSystem.Gradients.glass)
                    .overlay(
                        RoundedRectangle(cornerRadius: DesignSystem.CornerRadius.lg)
                            .stroke(Color.white.opacity(0.2), lineWidth: 1)
                    )
            )
    }
    
    func neumorphism(isPressed: Bool = false) -> some View {
        self
            .background(
                RoundedRectangle(cornerRadius: DesignSystem.CornerRadius.md)
                    .fill(DesignSystem.Colors.background)
                    .shadow(
                        color: isPressed ? .black.opacity(0.1) : .black.opacity(0.2),
                        radius: isPressed ? 2 : 8,
                        x: isPressed ? 1 : 4,
                        y: isPressed ? 1 : 4
                    )
                    .shadow(
                        color: isPressed ? .white.opacity(0.1) : .white.opacity(0.7),
                        radius: isPressed ? 1 : 8,
                        x: isPressed ? -1 : -4,
                        y: isPressed ? -1 : -4
                    )
            )
    }
    
    func cardStyle() -> some View {
        self
            .background(
                RoundedRectangle(cornerRadius: DesignSystem.CornerRadius.lg)
                    .fill(DesignSystem.Gradients.card)
                    .shadow(
                        color: .black.opacity(0.1),
                        radius: 8,
                        x: 0,
                        y: 4
                    )
            )
    }
}
```
